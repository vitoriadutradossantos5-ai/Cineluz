import os from gtts import gTTS from moviepy.editor import ImageClip, AudioFileClip, concatenate_videoclips def criar_video_de_dados(dados, arquivo_saida="video_gerado.mp4"): """ Gera um vídeo combinando textos convertidos em voz e imagens. :param dados: Lista de dicionários contendo 'texto' e 'imagem'. :param arquivo_saida: Nome do arquivo de vídeo final. """ clips_video = arquivos_temporarios = [] print("Iniciando a geração do vídeo...") try: for i, item in enumerate(dados): texto = item["texto"] imagem_path = item["imagem"] print(f"Processando cena {i+1} de {len(dados)}...") # 1. Gerar áudio a partir do texto (Text-to-Speech) audio_path = f"temp_audio_{i}.mp3" arquivos_temporarios.append(audio_path) tts = gTTS(text=texto, lang='pt', tld='com.br') tts.save(audio_path) # 2. Carregar o áudio e obter sua duração exata audio_clip = AudioFileClip(audio_path) duracao = audio_clip.duration # 3. Criar o clipe de imagem com a mesma duração do áudio image_clip = ImageClip(imagem_path).set_duration(duracao) # 4. Unir a imagem ao áudio correspondente clipe_cena = image_clip.set_audio(audio_clip) clips_video.append(clipe_cena) print("Unindo todas as cenas em um único vídeo...") # 5. Concatenar todos os clipes em um arquivo final video_final = concatenate_videoclips(clips_video, method="compose") video_final.write_videofile( arquivo_saida, fps=24, codec='libx264', audio_codec='aac', preset='medium' ) print(f"Sucesso! Vídeo salvo como: {arquivo_saida}") finally: # Limpar arquivos de áudio temporários print("Limpando arquivos temporários...") for arq in arquivos_temporarios: if os.path.exists(arq): os.remove(arq) if __name__ == "__main__": # Exemplo de dados estruturados (você pode carregar isso de um JSON, CSV ou Banco de Dados) dados_exemplo = [ { "texto": "Olá!
